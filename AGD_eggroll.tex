    EGGROLL update:
    \begin{align*}
        \tilde g(x) = \frac 1r \sum_{i=1}^r \frac{f(x+\sigma p_i) - f(x-\sigma p_i)}{2\sigma} p_i
    \end{align*}
    
    Nesterov AGD with parameter $\theta$:
    \begin{align*}
        x_{t} &= y_{t-1} - \eta \tilde g(y_{t-1}) \quad &\text{(progress)}\\
        y_t &= x_t + (1-\theta)(x_t - x_{t-1}) \quad &\text{(momentum)}
    \end{align*}
    Define Hamiltonian
    \begin{align*}
        E_t = \sqrt{\eta} f(x_{t}) + \frac{1}{2\sqrt{\eta}} \mg{v_t}^2
    \end{align*}
    where $v_t = x_t - x_{t-1}$ is the velocity at time t. 

    It follows that
    \begin{align*}
        f(x_{t+1}) + \frac{1}{2\eta} \mg{v_{t+1}}^2 
        \leq\;& f(x_t) + \langle \grad f(y_t), y_t-x_t \rangle 
        -\eta \langle \grad f(y_t), \tilde g (y_t) \rangle 
        \\& + \frac{L\eta^2}{2} \mg{\tilde g(y_t)}^2 
        + \frac{(1-\theta)^2}{\eta} \mg{x_t-x_{t-1}}^2 \\
        &-\langle \tilde g(y_t), y_t-x_t \rangle 
        + \frac{\eta}{2} \mg{\tilde g(y_t)}^2 .
    \end{align*}
    The term we care most about is
    \begin{align*}
        \grad f(y_t)^\top (I-pp^\top) (y_t-x_t)
    \end{align*}

    Run EGGROLL with $\tau = t_f(\delta)$ many steps. Then with probability at least $1-\delta$, for any starting point $x_0$, we have
    \begin{align*}
        f(x_\tau) - f(x_0) \leq -\frac{\eta}{4} \min_{0 \leq t \leq t_f(\delta)} \mg{\grad f(X_t)}^2 \leq -\frac{\eta}{4} \mg{\grad f(X_0)}^2
    \end{align*}
    Now, define $h \coloneqq x_\tau - x_0$, we will show this has good alignment with the gradient. Observe:
    \begin{align*}
        \mg{x_\tau - x_0} = \eta \mg{\sum_{t=1}^\tau \frac 1r \sum_{i=1}^r \langle \grad f(x_i), P_{t,j} \rangle P_{t,j}} \leq 4 \tau \eta \sqrt{\frac{dm}{r}} \mg{\grad f(x_0)}|
    \end{align*}
    Therefore,
    \begin{align*}
        \frac{\langle \grad f(x_0), -h \rangle}{\mg{\grad f(x_0)} \mg{h}} \geq \frac{1}{16\tau} \sqrt{\frac{r}{dm}}
    \end{align*}
    Meaning, $-h$ is aligned decently with the gradient, and stepping towards $-h$ decreases function value.
